export class Noise {

    constructor(
        private readonly seed: number
    ) {}

    sample(
        x: number,
        z: number
    ): number {

        const value =
            Math.sin((x + this.seed) * 0.017) *
            Math.cos((z - this.seed) * 0.019) +
            Math.sin((x + z) * 0.006);

        return value;
    }

}

export class OreGenerator {

    generate(chunk: Chunk): void {

        for (let i = 0; i < chunk.blocks.length; i++) {

            if (chunk.blocks[i] == 2) {

                const roll = Math.random();

                if (roll < 0.002)
                    chunk.blocks[i] = OreBlock.Diamond;

                else if (roll < 0.006)
                    chunk.blocks[i] = OreBlock.Gold;

                else if (roll < 0.02)
                    chunk.blocks[i] = OreBlock.Iron;

                else if (roll < 0.05)
                    chunk.blocks[i] = OreBlock.Coal;

            }

        }

    }

}

function createVoxelWorld(

  scene: Scene

) {

  const materials =

    createBlockMaterials(scene);



  const chunkSize = 16;



  const renderDistance = 2;



  for (

    let chunkX = -renderDistance;

    chunkX <= renderDistance;

    chunkX++

  ) {

    for (

      let chunkZ = -renderDistance;

      chunkZ <= renderDistance;

      chunkZ++

    ) {

      generateChunk(

        scene,

        materials,

        chunkX,

        chunkZ,

        chunkSize

      );

    }

  }

}



function generateChunk(

  scene: Scene,

  materials: Record<

    BlockType,

    StandardMaterial

  >,

  chunkX: number,

  chunkZ: number,

  chunkSize: number

) {

  for (

    let x = 0;

    x < chunkSize;

    x++

  ) {

    for (

      let z = 0;

      z < chunkSize;

      z++

    ) {

      const worldX =

        chunkX *

          chunkSize +

        x;



      const worldZ =

        chunkZ *

          chunkSize +

        z;



      const height =

        getTerrainHeight(

          worldX,

          worldZ

        );



      for (

        let y = 0;

        y <= height;

        y++

      ) {

        let type: BlockType;



        if (

          y === height

        ) {

          type = "grass";

        } else if (

          y > height - 4

        ) {

          type = "dirt";

        } else {

          type = "stone";

        }



        const block =

          MeshBuilder.CreateBox(

            "voxel",

            {

              size: 1

            },

            scene

          );



        block.position =

          new Vector3(

            worldX,

            y,

            worldZ

          );



        block.material =

          materials[type];



        block.checkCollisions =

          true;

      }

    }

  }

}



function getTerrainHeight(

  x: number,

  z: number

) {

  const hills =

    Math.sin(

      x * 0.12

    ) * 3;



  const mountains =

    Math.cos(

      z * 0.08

    ) * 4;



  const detail =

    Math.sin(

      (x + z) * 0.25

    );



  return Math.max(

    1,

    Math.floor(

      5 +

      hills +

      mountains +

      detail

    )

  );

}



function createBlockMaterials(

  scene: Scene

) {

  const grass =

    new StandardMaterial(

      "grass",

      scene

    );



  grass.diffuseColor =

    new Color3(

      0.15,

      0.55,

      0.2

    );



  const dirt =

    new StandardMaterial(

      "dirt",

      scene

    );



  dirt.diffuseColor =

    new Color3(

      0.35,

      0.18,

      0.08

    );



  const stone =

    new StandardMaterial(

      "stone",

      scene

    );



  stone.diffuseColor =

    new Color3(

      0.35,

      0.38,

      0.42

    );



  return {

    air: grass,

    grass,

    dirt,

    stone

  };

}

import {

  BlockID,

  BLOCKS

} from "./BlockRegistry";



export const CHUNK_SIZE = 16;



export const CHUNK_HEIGHT = 128;



export class Chunk {

  readonly x: number;



  readonly z: number;



  private blocks: BlockID[];



  constructor(

    x: number,

    z: number

  ) {

    this.x = x;



    this.z = z;



    this.blocks =

      new Array(

        CHUNK_SIZE *

          CHUNK_HEIGHT *

          CHUNK_SIZE

      ).fill("air");

  }



  private index(

    x: number,

    y: number,

    z: number

  ) {

    return (

      x +

      CHUNK_SIZE *

        (z +

          CHUNK_SIZE *

            y)

    );

  }



  getBlock(

    x: number,

    y: number,

    z: number

  ): BlockID {

    if (

      x < 0 ||

      x >= CHUNK_SIZE ||

      z < 0 ||

      z >= CHUNK_SIZE ||

      y < 0 ||

      y >= CHUNK_HEIGHT

    ) {

      return "air";

    }



    return this.blocks[

      this.index(x, y, z)

    ];

  }



  setBlock(

    x: number,

    y: number,

    z: number,

    block: BlockID

  ) {

    if (

      x < 0 ||

      x >= CHUNK_SIZE ||

      z < 0 ||

      z >= CHUNK_SIZE ||

      y < 0 ||

      y >= CHUNK_HEIGHT

    ) {

      return;

    }



    this.blocks[

      this.index(x, y, z)

    ] = block;

  }



  isSolid(

    x: number,

    y: number,

    z: number

  ) {

    const block =

      this.getBlock(

        x,

        y,

        z

      );



    return BLOCKS[

      block

    ].solid;

  }

}
